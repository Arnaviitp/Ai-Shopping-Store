# Quick Node proxy (run locally, then keep this file and open in browser at http://localhost:3000/index.html)
npm init -y
npm i express node-fetch
// save as server.js:
const express = require('express'); const fetch = (...a)=>import('node-fetch').then(m=>m.default(...a));
const app = express(); app.use(express.json()); app.use(express.static('.'));
const GROQ_KEY = process.env.GROQ_KEY || '';
app.post('/proxy', async (req,res)=>{ try{
  const r = await fetch('https://api.groq.ai/v1/chat/completions',{method:'POST',headers:{'Content-Type':'application/json','Authorization':'Bearer '+GROQ_KEY},body:JSON.stringify(req.body)});
  const j = await r.text(); res.status(r.status).send(j);
}catch(e){res.status(500).send(e.message);} });
app.listen(3000,()=>console.log('Proxy http://localhost:3000')); 
// Run: GROQ_KEY="gsk_..." node server.js
